# ============================================================
# CPU
# ============================================================

CPU_USAGE_PERCENT =
100 - (
  avg by (instance) (
    rate(
      node_cpu_seconds_total{
        job="node-exporter",
        mode="idle"
      }[1m]
    )
  ) * 100
)


CPU_USAGE_BY_CORE_PERCENT =
100 - (
  avg by (instance, cpu) (
    rate(
      node_cpu_seconds_total{
        job="node-exporter",
        mode="idle"
      }[1m]
    )
  ) * 100
)


CPU_USAGE_BY_MODE_PERCENT =

    node_cpu_seconds_total{

  )

# MEMORY
# ============================================================

MEMORY_USAGE_PERCENT =
(
  1 -
  (
    node_memory_MemAvailable_bytes{
      job="node-exporter"
    }
    /
    node_memory_MemTotal_bytes{
      job="node-exporter"
    }
  )
) * 100


MEMORY_AVAILABLE_PERCENT =
(
  node_memory_MemAvailable_bytes{
    job="node-exporter"
  }
  /
  node_memory_MemTotal_bytes{
    job="node-exporter"
  }
) * 100

MEMORY_BUFFERS_BYTES =

SWAP_USAGE_PERCENT_SAFE =
(
  (
    node_memory_SwapTotal_bytes{
      job="node-exporter"
    }
    -
    node_memory_SwapFree_bytes{
      job="node-exporter"
    }
  )
  /
  clamp_min(
    node_memory_SwapTotal_bytes{
      job="node-exporter"
    },
    1
  )
) * 100

# ============================================================
# COMPOSITE PRESSURE INDEX
# Higher value = more resource pressure
# ============================================================

RESOURCE_PRESSURE_INDEX =
(
  (
    100 -
    (
      avg by (instance) (
        rate(
          node_cpu_seconds_total{
            job="node-exporter",
            mode="idle"
          }[1m]
        )
      ) * 100
    )
  ) * 0.4
)
+
(
  (
    1 -
    (
      node_memory_MemAvailable_bytes{
        job="node-exporter"
      }
      /
      node_memory_MemTotal_bytes{
        job="node-exporter"
      }
    )
  ) * 100 * 0.4
)
+
(
  (
    100 *
    (
      1 -
      (
        node_filesystem_avail_bytes{
          job="node-exporter",
          mountpoint="/",
          fstype!~"tmpfs|overlay|squashfs|nsfs|ramfs"
        }
        /
        node_filesystem_size_bytes{
          job="node-exporter",
          mountpoint="/",
          fstype!~"tmpfs|overlay|squashfs|nsfs|ramfs"
        }
      )
    )
  ) * 0.2
)
